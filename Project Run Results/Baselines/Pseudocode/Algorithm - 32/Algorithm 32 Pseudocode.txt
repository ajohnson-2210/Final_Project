FUNCTION substring_range ACCEPTS(text, start_index, end_index)
    result <- EMPTY STRING
    index <- start_index
    WHILE index < end_index DO
        result <- result + text[index]
        index <- index + 1
    ENDWHILE
    RETURN result
ENDFUNCTION

FUNCTION sublist_range ACCEPTS(values, start_index, end_index)
    result <- EMPTY LIST
    index <- start_index
    WHILE index < end_index DO
        APPEND values[index] TO result
        index <- index + 1
    ENDWHILE
    RETURN result
ENDFUNCTION

FUNCTION clean_text ACCEPTS(t)
    t <- t.replace(STRING("’"), STRING("'"))
    t <- t.replace(STRING("‘"), STRING("'"))
    t <- t.replace(STRING("“"), STRING("\""))
    t <- t.replace(STRING("”"), STRING("\""))
    t <- t.replace(STRING("—"), STRING("-"))
    t <- t.replace(STRING("–"), STRING("-"))
    t <- t.replace(STRING("…"), STRING("..."))
    RETURN t
ENDFUNCTION

FUNCTION split_sentences ACCEPTS(text)
    temp <- text.replace(STRING("?"), STRING(".")).replace(STRING("!"), STRING("."))
    parts <- temp.split(STRING("."))
    sentences <- EMPTY LIST
    i <- 0
    loop <- BOOLEAN(TRUE)
    WHILE loop DO
        TRY
            s <- parts[i].strip()
            IF LENGTH OF s > 0 THEN
                APPEND s TO sentences
            ENDIF
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE
    RETURN sentences
ENDFUNCTION

FUNCTION words_match ACCEPTS(w1, w2)
    IF w1 == w2 THEN
        RETURN BOOLEAN(TRUE)
    ENDIF
    i <- 0
    loop <- BOOLEAN(TRUE)
    WHILE loop DO
        TRY
            synonym_pair <- synonyms[i]
            a <- synonym_pair[0]
            b <- synonym_pair[1]
            IF w1 == a AND w2 == b OR w1 == b AND w2 == a THEN
                RETURN BOOLEAN(TRUE)
            ENDIF
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE
    RETURN BOOLEAN(FALSE)
ENDFUNCTION

                    WHILE loop2 DO
                        TRY
                            IF q_words[i2] NOT IN stopwords THEN
                                obj <- q_words[i2]
                            ENDIF
                            i2 <- i2 + 1
                        CATCH
                            loop2 <- BOOLEAN(FALSE)
                        ENDTRY
                    ENDWHILE
                    s2 <- 0
                    loop2 <- BOOLEAN(TRUE)
                    WHILE loop2 DO
                        TRY
                            IF s_words[s2] == subject THEN
                                score <- score + 3
                            ENDIF
                            IF s_words[s2] == obj THEN
                                score <- score + 4
                            ENDIF
                            s2 <- s2 + 1
                        CATCH
                            loop2 <- BOOLEAN(FALSE)
                        ENDTRY
                    ENDWHILE
                    s2 <- 0
                    loop2 <- BOOLEAN(TRUE)
                    WHILE loop2 DO
                        TRY
                            IF s_words[s2] IN LIST [STRING("was"), STRING("were"), STRING("had"), STRING("approached"), STRING("contained"), STRING("included")] THEN
                                score <- score + 2
                            ENDIF
                            s2 <- s2 + 1
                        CATCH
                            loop2 <- BOOLEAN(FALSE)
                        ENDTRY
                    ENDWHILE
                ENDIF
                s <- s + 1
            CATCH
                s_loop <- BOOLEAN(FALSE)
            ENDTRY
        ENDWHILE
        IF i > 0 THEN
            prev_words <- CALL tokenise(sentences[i - 1])
            q <- 0
            q_loop <- BOOLEAN(TRUE)
            WHILE q_loop DO
                TRY
                    qw <- q_words[q]
                    IF qw NOT IN stopwords THEN
                        p <- 0
                        p_loop <- BOOLEAN(TRUE)
                        WHILE p_loop DO
                            TRY
                                IF CALL words_match(prev_words[p], qw) THEN
                                    score <- score + 1
                                ENDIF
                                p <- p + 1
                            CATCH
                                p_loop <- BOOLEAN(FALSE)
                            ENDTRY
                        ENDWHILE
                    ENDIF
                    q <- q + 1
                CATCH
                    q_loop <- BOOLEAN(FALSE)
                ENDTRY
            ENDWHILE
        ENDIF
        IF i < (LENGTH OF sentences - 1) THEN
            next_words <- CALL tokenise(sentences[i + 1])
            q <- 0
            q_loop <- BOOLEAN(TRUE)
            WHILE q_loop DO
                TRY
                    qw <- q_words[q]
                    IF qw NOT IN stopwords THEN
                        n2 <- 0
                        n2_loop <- BOOLEAN(TRUE)
                        WHILE n2_loop DO
                            TRY
                                IF CALL words_match(next_words[n2], qw) THEN
                                    score <- score + 1
                                ENDIF
                                n2 <- n2 + 1
                            CATCH
                                n2_loop <- BOOLEAN(FALSE)
                            ENDTRY
                        ENDWHILE
                    ENDIF
                    q <- q + 1
                CATCH
                    q_loop <- BOOLEAN(FALSE)
                ENDTRY
            ENDWHILE
        ENDIF
        APPEND TUPLE(score, sent) TO scores
        i <- i + 1
    CATCH
        loop <- BOOLEAN(FALSE)
    ENDTRY
ENDWHILE
best_score <- -1
best_sentence <- NONE

FUNCTION rewrite_answer ACCEPTS(sentence, question_type)
    words <- CALL tokenise(sentence)
    IF LENGTH OF words > 0 THEN
        IF words[0] == STRING("because") THEN
            words <- CALL sublist_range(words, 1, LENGTH OF words)
        ENDIF
    ENDIF
    IF question_type == STRING("where") THEN
        i <- 0
        loop <- BOOLEAN(TRUE)
        WHILE loop DO
            TRY
                IF words[i] == STRING("at") THEN
                    words[i] <- STRING("in")
                ENDIF
                i <- i + 1
            CATCH
                loop <- BOOLEAN(FALSE)
            ENDTRY
        ENDWHILE
    ENDIF
    IF question_type == STRING("why") THEN
        i <- 0
        loop <- BOOLEAN(TRUE)
        start <- 0
        WHILE loop DO
            TRY
                IF words[i] == STRING("because") THEN
                    start <- i
                ENDIF
                i <- i + 1
            CATCH
                loop <- BOOLEAN(FALSE)
            ENDTRY
        ENDWHILE
        words <- CALL sublist_range(words, start, LENGTH OF words)
    ENDIF
    IF question_type == STRING("who") THEN
        IF LENGTH OF words > 8 THEN
            new_words <- EMPTY LIST
            j <- 0
            loop2 <- BOOLEAN(TRUE)
            WHILE loop2 DO
                TRY
                    IF j < 8 THEN
                        APPEND words[j] TO new_words
                    ENDIF
                    j <- j + 1
                CATCH
                    loop2 <- BOOLEAN(FALSE)
                ENDTRY
            ENDWHILE
            words <- new_words
        ENDIF
    ENDIF
    IF question_type == STRING("what") THEN
        i <- 0
        loop <- BOOLEAN(TRUE)
        start <- 0
        found <- BOOLEAN(FALSE)
        WHILE loop DO
            TRY
                IF words[i] == STRING("for") THEN
                    start <- i
                    found <- BOOLEAN(TRUE)
                ENDIF
                i <- i + 1
            CATCH
                loop <- BOOLEAN(FALSE)
            ENDTRY
        ENDWHILE
        IF found THEN
            new_words <- EMPTY LIST
            j <- start
            loop2 <- BOOLEAN(TRUE)
            WHILE loop2 DO
                TRY
                    APPEND words[j] TO new_words
                    j <- j + 1
                CATCH
                    loop2 <- BOOLEAN(FALSE)
                ENDTRY
            ENDWHILE
            words <- new_words
        ENDIF
    ENDIF
    rewritten <- EMPTY STRING
    i <- 0
    loop <- BOOLEAN(TRUE)
    WHILE loop DO
        TRY
            rewritten <- rewritten + (words[i] + STRING(" "))
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE
    RETURN rewritten.strip()
ENDFUNCTION

IF best_sentence != NONE AND best_score > 0 THEN
    answer <- CALL rewrite_answer(best_sentence, question_type)
ENDIF
