Solve the N Queens Problem for an arbitrary board size N


def safe_placement(board, row, col, n):
    r = 0
    while r < row:
        if board[r] == col:
            return False
        r += 1

    r = row - 1
    c = col - 1
    while r >= 0 and c >= 0:
        if board[r] == c:
            return False
        r -= 1
        c -= 1

    r = row - 1
    c = col + 1
    while r >= 0 and c < n:
        if board[r] == c:
            return False
        r -= 1
        c += 1

    return True


def solve_n_queens_util(board, row, n, solutions):
    if row == n:
        solutions.append(board[:])
        return

    col = 0
    while col < n:
        if safe_placement(board, row, col, n):
            board[row] = col
            solve_n_queens_util(board, row + 1, n, solutions)
            board[row] = -1
        col += 1


def solve_n_queens(n):
    board = [-1] * n
    solutions = []
    solve_n_queens_util(board, 0, n, solutions)
    return solutions


def print_board(solution):
    n = len(solution)
    r = 0
    while r < n:
        row_str = ""
        c = 0
        while c < n:
            if solution[r] == c:
                row_str += "Q "
            else:
                row_str += ". "
            c += 1
        print(row_str)
        r += 1
    print("")


def print_all_solutions(solutions):
    total = len(solutions)
    print(f"\nShowing ALL {total} solutions:\n")
    i = 0
    while i < total:
        print(f"Solution {i + 1}:")
        print_board(solutions[i])
        i += 1

# MAIN PROGRAM FLOW
# ============================================================

# 1. Input N
N = int(input("Enter board size N: "))

# 2. Solve and store solutions
solutions = solve_n_queens(N)
total = len(solutions)

print(f"\nSolved {N}-Queens.")
print(f"Total solutions: {total}\n")

# 3. Display menu loop
index = 0

while True:
    print("Menu:")
    print("  1 - Show current solution")
    print("  2 - Next solution")
    print("  3 - Previous solution")
    print("  4 - Go to solution number")
    print("  5 - Show ALL solutions")
    print("  6 - Quit")

    choice = input("Enter choice: ").strip()

    # 4. Display results based on choice
    if choice == "1":
        if total > 0:
            print(f"\nShowing solution {index + 1} of {total}:\n")
            print_board(solutions[index])
        else:
            print("\nNo solutions exist for this N.\n")

    elif choice == "2":
        if total > 0:
            index = (index + 1) % total
            print(f"\nShowing solution {index + 1} of {total}:\n")
            print_board(solutions[index])
        else:
            print("\nNo solutions.\n")

    elif choice == "3":
        if total > 0:
            index = (index - 1) % total
            print(f"\nShowing solution {index + 1} of {total}:\n")
            print_board(solutions[index])
        else:
            print("\nNo solutions.\n")

    elif choice == "4":
        if total > 0:
            num = input("Enter solution number: ").strip()
            if num.isdigit():
                num = int(num)
                if 1 <= num <= total:
                    index = num - 1
                    print(f"\nShowing solution {index + 1} of {total}:\n")
                    print_board(solutions[index])
                else:
                    print("\nInvalid number.\n")
            else:
                print("\nInvalid input.\n")
        else:
            print("\nNo solutions.\n")

    elif choice == "5":
        print_all_solutions(solutions)

    elif choice == "6":
        print("\nExiting program.\n")
        break

    else:
        print("\nInvalid option.\n")
