An ordered list of numbers was rotated at an unknown pivot point (e.g. [4,5,6,1,2,3]). Find the index of a target value in this list in logarithmic time

        
print(f"Binary Search in O(log n) Time")

def rsearch(nums, target):
    left = 0
    right = len(nums) - 1

    while left <= right:
        mid = (left + right) // 2
        mid_val = nums[mid]

        if mid_val == target:
            return mid

        # Left half is sorted
        if nums[left] <= mid_val:
            if nums[left] <= target < mid_val:
                right = mid - 1
            else:
                left = mid + 1
        # Right half is sorted
        else:
            if mid_val < target <= nums[right]:
                left = mid + 1
            else:
                right = mid - 1

    return "Target Value Does Not Exist"
   
nums = [4, 5, 6, 1, 2, 3]
print(nums)
print(rsearch(nums, 1))
print(rsearch(nums, 5))
print(rsearch(nums, 9))
