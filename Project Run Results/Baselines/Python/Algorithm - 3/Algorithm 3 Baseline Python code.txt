Determine if a year satisfies the leap year rules

leap = False
loop = True
while loop == True:
    year = input("Enter Year: ")
    try:
        year = int(year);
        loop = False
    except:
        print("That's Not a Valid Year, Please Try Again!")
        
if year % 4 == 0:
    leap = True
else:
    leap = False
    
if year % 100 == 0 and not year % 400 == 0:
    leap = False
elif year % 400 == 0:
    leap = True
    
if not leap:
    print(f"{year} is Not a Leap Year!")
else:
    print(f"{year} Is a Leap Year!")
