def clean_text(t):
    t = t.replace("’", "'")
    t = t.replace("‘", "'")
    t = t.replace("“", '"')
    t = t.replace("”", '"')
    t = t.replace("—", "-")
    t = t.replace("–", "-")
    t = t.replace("…", "...")
    return t

def split_sentences(text):
    temp = text.replace("?", ".").replace("!", ".")
    parts = temp.split(".")
    sentences = []
    i = 0
    loop = True
    while loop:
        try:
            s = parts[i].strip()
            if len(s) > 0:
                sentences.append(s)
            i += 1
        except:
            loop = False
    return sentences
