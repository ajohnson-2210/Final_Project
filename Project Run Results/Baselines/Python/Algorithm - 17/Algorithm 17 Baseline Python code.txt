Computer the sum and average of averages list of numbers

numlist = [7,9,3,5,2.6,8.3]
total = 0
counter = 0
while len(numlist) > 0:
    total += numlist.pop(0)
    counter += 1
average = total / counter
total = format(total, '.2f')
average = format(average, '.2f')
	
print(f"The Sum of Numbers in the list = {total}")
print(f"The Average = {average}")
