FUNCTION levelOrderTraversal(root):
    IF root IS NULL:
        RETURN empty list
    result = empty list
    queue = empty queue
    queue.enqueue(root)
    WHILE queue IS NOT empty:
        levelSize = queue.size()
        currentLevel = empty list
        FOR i FROM 0 TO levelSize-1:
            node = queue.dequeue()
            currentLevel.append(node.value)
            IF node.left != NULL:
                queue.enqueue(node.left)
            IF node.right != NULL:
                queue.enqueue(node.right)
        result.append(currentLevel)
    RETURN result

FUNCTION slidingWindowMax(arr, k):
    result = empty list
    deque = empty deque
    FOR i FROM 0 TO length(arr)-1:
        // Remove elements out of window from front
        WHILE deque IS NOT empty AND deque.front() <= i - k:
            deque.pop_front()
        // Maintain decreasing order in deque
        WHILE deque IS NOT empty AND arr[deque.back()] <= arr[i]:
            deque.pop_back()
        deque.push_back(i)
        // Add max to result once window is full
        IF i >= k - 1:
            result.append(arr[deque.front()])
    RETURN result

FUNCTION integerSqrt(n):
    IF n < 2:
        RETURN n
    low = 1
    high = n
    ans = 0
    WHILE low <= high:
        mid = (low + high) DIV 2
        square = mid * mid
        IF square == n:
            RETURN mid
        ELSE IF square < n:
            ans = mid
            low = mid + 1
        ELSE:
            high = mid - 1
    RETURN ans

35. Design a memory buffer that holds a fixed number of items. When it runs out of space, it must automatically discard the item that hasn't been accessed for the longest time. (LRU Cache)

CLASS LRUCache:
    // Use a combination of a dictionary and a doubly linked list (or an ordered map)
    cache = empty dictionary
    capacity = given capacity
    order = empty list (or a doubly linked list to maintain access order)

    FUNCTION get(key):
        IF key NOT IN cache:
            RETURN -1 // or NULL
        // Move key to front of order (most recently used)
        order.remove(key)
        order.insert_front(key)
        RETURN cache[key]

    FUNCTION put(key, value):
        IF key IN cache:
            // Update value and move key to front
            cache[key] = value
            order.remove(key)
            order.insert_front(key)
        ELSE:
            IF length(order) >= capacity:
                // Evict the least recently used (last element in order)
                lruKey = order.back()
                order.remove_back()
                cache.remove(lruKey)
            order.insert_front(key)
            cache[key] = value

FUNCTION searchRotatedArray(arr, target):
    low = 0
    high = length(arr) - 1
    WHILE low <= high:
        mid = (low + high) DIV 2
        IF arr[mid] == target:
            RETURN mid
        // Check if left half is sorted
        IF arr[low] <= arr[mid]:
            IF arr[low] <= target AND target <= arr[mid]:
                high = mid - 1
            ELSE:
                low = mid + 1
        // Right half must be sorted
        ELSE:
            IF arr[mid] <= target AND target <= arr[high]:
                low = mid + 1
            ELSE:
                high = mid - 1
    RETURN -1

CLASS Stack:
    stackArray = empty array
    topIndex = -1

    FUNCTION push(item):
        topIndex = topIndex + 1
        stackArray[topIndex] = item

    FUNCTION pop():
        IF topIndex < 0:
            RETURN NULL // Stack underflow
        item = stackArray[topIndex]
        topIndex = topIndex - 1
        RETURN item

    FUNCTION peek():
        IF topIndex < 0:
            RETURN NULL
        RETURN stackArray[topIndex]

41. Solve the N Queens Problem for N arbitrary board size N.

text
FUNCTION solveNQueens(N):
    board = 2D array of size NxN initialized to 0
    solutions = empty list

    FUNCTION isSafe(board, row, col):
        // Check column
        FOR i FROM 0 TO row-1:
            IF board[i][col] == 1:
                RETURN False
        // Check upper-left diagonal
        i = row - 1, j = col - 1
        WHILE i >= 0 AND j >= 0:
            IF board[i][j] == 1:
                RETURN False
            i = i - 1, j = j - 1
        // Check upper-right diagonal
        i = row - 1, j = col + 1
        WHILE i >= 0 AND j < N:
            IF board[i][j] == 1:
                RETURN False
            i = i - 1, j = j + 1
        RETURN True

    FUNCTION placeQueen(board, row):
        IF row == N:
            solutions.append(copy of board)
            RETURN
        FOR col FROM 0 TO N-1:
            IF isSafe(board, row, col):
                board[row][col] = 1
                placeQueen(board, row + 1)
                board[row][col] = 0 // Backtrack

    placeQueen(board, 0)
    RETURN solutions

49. Create a pathfinding solution that guides a user through a maze using graph traversal (BFS).

FUNCTION findPath(maze, start, end):
    // BFS traversal
    queue = empty queue
    visited = 2D boolean array same size as maze initialized to False
    parent = 2D array to store previous cell for reconstruction

    queue.enqueue(start)
    visited[start.row][start.col] = True

    directions = [(0,1), (1,0), (0,-1), (-1,0)] // right, down, left, up

    WHILE queue IS NOT empty:
        current = queue.dequeue()
        IF current == end:
            // Reconstruct path
            path = empty list
            WHILE current != start:
                path.append(current)
                current = parent[current.row][current.col]
            path.append(start)
            path.reverse()
            RETURN path
        FOR each direction in directions:
            newRow = current.row + direction.row
            newCol = current.col + direction.col
            IF newRow, newCol are within bounds AND maze[newRow][newCol] is not blocked AND visited[newRow][newCol] == False:
                visited[newRow][newCol] = True
                parent[newRow][newCol] = current
                queue.enqueue((newRow, newCol))
    RETURN NULL // No path found
