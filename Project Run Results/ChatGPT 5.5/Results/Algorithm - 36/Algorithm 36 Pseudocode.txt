# Find a target element in a rotated sorted array.
BEGIN
    INPUT Array, Target

    Low ← 0
    High ← LENGTH(Array) - 1

    WHILE Low ≤ High
        Mid ← (Low + High) DIV 2

        IF Array[Mid] = Target THEN
            OUTPUT Mid
            STOP
        ENDIF

        IF Array[Low] ≤ Array[Mid] THEN
            IF Target ≥ Array[Low] AND Target < Array[Mid] THEN
                High ← Mid - 1
            ELSE
                Low ← Mid + 1
            ENDIF
        ELSE
            IF Target > Array[Mid] AND Target ≤ Array[High] THEN
                Low ← Mid + 1
            ELSE
                High ← Mid - 1
            ENDIF
        ENDIF
    ENDWHILE

    OUTPUT "Not Found"
END
