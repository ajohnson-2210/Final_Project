# Develop a process for multiplying two matrices.
BEGIN
    INPUT MatrixA, MatrixB
    INPUT RowsA, ColumnsA, ColumnsB

    FOR i ← 0 TO RowsA-1
        FOR j ← 0 TO ColumnsB-1
            Result[i][j] ← 0

            FOR k ← 0 TO ColumnsA-1
                Result[i][j] ← Result[i][j] +
                                MatrixA[i][k] × MatrixB[k][j]
            ENDFOR
        ENDFOR
    ENDFOR

    OUTPUT Result
END
