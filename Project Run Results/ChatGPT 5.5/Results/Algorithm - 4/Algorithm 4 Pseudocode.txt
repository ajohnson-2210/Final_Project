# Find the intersection of two sorted arrays.
BEGIN
    INPUT Array1, Array2

    i ← 0
    j ← 0

    WHILE i < LENGTH(Array1) AND j < LENGTH(Array2)
        IF Array1[i] = Array2[j] THEN
            OUTPUT Array1[i]
            i ← i + 1
            j ← j + 1
        ELSE IF Array1[i] < Array2[j] THEN
            i ← i + 1
        ELSE
            j ← j + 1
        ENDIF
    ENDWHILE
END
