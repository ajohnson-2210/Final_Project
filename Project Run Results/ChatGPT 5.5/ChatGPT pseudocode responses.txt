1. Determine whether two given strings are anagrams of each other.
BEGIN
    INPUT String1, String2

    IF LENGTH(String1) ≠ LENGTH(String2) THEN
        OUTPUT "Not Anagrams"
    ELSE
        SORT characters of String1
        SORT characters of String2

        IF String1 = String2 THEN
            OUTPUT "Anagrams"
        ELSE
            OUTPUT "Not Anagrams"
        ENDIF
    ENDIF
END

2. Convert a non-negative decimal integer into its binary string representation.
BEGIN
    INPUT Number
    Binary ← ""

    IF Number = 0 THEN
        OUTPUT "0"
    ELSE
        WHILE Number > 0
            Remainder ← Number MOD 2
            Binary ← Remainder + Binary
            Number ← Number DIV 2
        ENDWHILE

        OUTPUT Binary
    ENDIF
END

33. Find the square root of a non-negative integer using binary search.
BEGIN
    INPUT Number

    Low ← 0
    High ← Number
    Answer ← 0

    WHILE Low ≤ High
        Mid ← (Low + High) DIV 2

        IF Mid × Mid = Number THEN
            OUTPUT Mid
            STOP
        ELSE IF Mid × Mid < Number THEN
            Answer ← Mid
            Low ← Mid + 1
        ELSE
            High ← Mid - 1
        ENDIF
    ENDWHILE

    OUTPUT Answer
END

BEGIN

    Slow ← Head

        Slow ← Slow.Next
        Fast ← Fast.Next.Next

BEGIN

        IF Row = N THEN
            OUTPUT Board
            RETURN
        ENDIF

        FOR Col ← 0 TO N-1
            IF Position(Row, Col) is safe THEN
                PLACE Queen at (Row, Col)

                Solve(Row + 1)

                REMOVE Queen from (Row, Col)
            ENDIF
        ENDFOR
    END FUNCTION

END

43. Solve the Tower of Hanoi problem for N discs.
BEGIN
    INPUT N

    FUNCTION Hanoi(Discs, Source, Auxiliary, Destination)
        IF Discs = 1 THEN
            OUTPUT "Move disk from", Source, "to", Destination
        ELSE
            Hanoi(Discs-1, Source, Destination, Auxiliary)

            OUTPUT "Move disk from", Source, "to", Destination

            Hanoi(Discs-1, Auxiliary, Source, Destination)
        ENDIF
    END FUNCTION

    Hanoi(N, "A", "B", "C")
END

44. Find the contiguous subarray with the maximum sum (Kadane's Algorithm).
BEGIN
    INPUT Array

    CurrentSum ← Array[0]
    MaxSum ← Array[0]

    FOR i ← 1 TO LENGTH(Array)-1
        CurrentSum ← MAX(Array[i], CurrentSum + Array[i])

        IF CurrentSum > MaxSum THEN
            MaxSum ← CurrentSum
        ENDIF
    ENDFOR

    OUTPUT MaxSum
END

49. Create a pathfinding solution through a maze using graph traversal.
BEGIN
    INPUT Maze
    INPUT Start, Goal

    CREATE Queue
    MARK Start as visited
    ENQUEUE Start

    WHILE Queue is not empty
        Current ← DEQUEUE Queue

        IF Current = Goal THEN
            OUTPUT "Path Found"
            STOP
        ENDIF

        FOR EACH Neighbour of Current
            IF Neighbour not visited THEN
                MARK Neighbour as visited
                ENQUEUE Neighbour
            ENDIF
        ENDFOR
    ENDWHILE

    OUTPUT "No Path Found"
END
