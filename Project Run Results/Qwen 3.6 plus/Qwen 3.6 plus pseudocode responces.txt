# Anagram Check

INPUT str1, str2

IF length(str1) != length(str2):
    RETURN FALSE

CREATE map charCounts

FOR each char in str1:
    charCounts[char]++

FOR each char in str2:
    charCounts[char]--

FOR each count in charCounts:
    IF count != 0:
        RETURN FALSE

RETURN TRUE

# Longest Substring Without Repeating Chars

INPUT s

maxLen = 0
start = 0
seen = {}

FOR i from 0 to length(s)-1:
    IF s[i] in seen AND seen[s[i]] >= start:
        start = seen[s[i]] + 1

    seen[s[i]] = i
    maxLen = MAX(maxLen, i - start + 1)

RETURN maxLen


# Binary Tree Level-Order Traversal

INPUT root

IF root == null:
    RETURN []

queue = [root]
result = []

WHILE queue not empty:
    levelSize = length(queue)
    currentLevel = []

    FOR _ from 0 to levelSize-1:
        node = queue.pop(0)
        currentLevel.append(node.value)

        IF node.left:
            queue.append(node.left)

        IF node.right:
            queue.append(node.right)

    result.append(currentLevel)

RETURN result


# Rotate Matrix 90° Clockwise

INPUT matrix (N x N)

FOR i from 0 to N/2 - 1:
    FOR j from i to N-1-i-1:
        temp = matrix[i][j]
        matrix[i][j] = matrix[N-1-j][i]
        matrix[N-1-j][i] = matrix[N-1-i][N-1-j]
        matrix[N-1-i][N-1-j] = matrix[j][N-1-i]
        matrix[j][N-1-i] = temp

RETURN matrix


# Sliding Window Maximum

INPUT arr, k

result = []
deque = []

FOR i from 0 to length(arr)-1:
    IF deque[0] == i - k:
        deque.popFront()

    WHILE deque not empty AND arr[deque.back()] <= arr[i]:
        deque.popBack()

    deque.pushBack(i)

    IF i >= k - 1:
        result.append(arr[deque.front()])

RETURN result

# Integer Square Root (Binary Search)

INPUT n

IF n < 2:
    RETURN n

left = 1
right = n DIV 2
ans = 1

WHILE left <= right:
    mid = (left + right) DIV 2

    IF mid * mid <= n:
        ans = mid
        left = mid + 1
    ELSE:
        right = mid - 1

RETURN ans

# LRU Memory Buffer

INIT capacity, cache = orderedDict()

FUNCTION get(key):
    IF key in cache:
        moveToFront(cache, key)
        RETURN cache[key]
    ELSE:
        RETURN null

FUNCTION put(key, value):
    IF key in cache:
        moveToFront(cache, key)
        cache[key] = value
    ELSE:
        IF length(cache) == capacity:
            removeBack(cache)

        addToFront(cache, key, value)


# Search Rotated Sorted Array

INPUT arr, target

left = 0
right = length(arr)-1

WHILE left <= right:
    mid = (left + right) DIV 2

    IF arr[mid] == target:
        RETURN mid

    IF arr[left] <= arr[mid]:
        IF arr[left] <= target < arr[mid]:
            right = mid - 1
        ELSE:
            left = mid + 1
    ELSE:
        IF arr[mid] < target <= arr[right]:
            left = mid + 1
        ELSE:
            right = mid - 1

RETURN -1


# Stack Using Array

INIT stack = []
size = 0

FUNCTION push(x):
    stack.append(x)
    size++

FUNCTION pop():
    IF size > 0:
        size--
        RETURN stack.pop()
    ELSE:
        ERROR

FUNCTION peek():
    IF size > 0:
        RETURN stack[size-1]
    ELSE:
        ERROR

# N Queens Problem

INPUT N

FUNCTION placeQueen(row, board):
    IF row == N:
        RETURN TRUE

    FOR col from 0 to N-1:
        IF isSafe(board, row, col):
            board[row][col] = 'Q'

            IF placeQueen(row + 1, board):
                RETURN TRUE

            board[row][col] = '.'

    RETURN FALSE

RETURN placeQueen(0, emptyBoard(N))

# Recursive Folder Search

FUNCTION search(directory, targetExt):
    results = []

    FOR item in directory.contents:
        IF item is file AND item.ext == targetExt:
            results.add(item)
        ELSE IF item is directory:
            results.extend(search(item, targetExt))

    RETURN results


# Maze Pathfinding (BFS)

INPUT maze, start, end

queue = [(start, [start])]
visited = {start}

WHILE queue not empty:
    curr, path = queue.popFront()

    IF curr == end:
        RETURN path

    FOR neighbor in getValidNeighbors(curr, maze):
        IF neighbor not in visited:
            visited.add(neighbor)
            queue.append((neighbor, path + [neighbor]))

RETURN "No Path"
