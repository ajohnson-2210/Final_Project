# An ordered list of numbers was rotated at an unknown pivot point (e.g., [4,5,6,1,2,3]). Find the index of a target element in this list in logarithmic time
FUNCTION FIND_TARGET(rotated_list, target)
  low = 0
  high = LENGTH(rotated_list) - 1
  WHILE low <= high
    mid = (low + high) / 2
    IF rotated_list[mid] == target THEN
      RETURN mid
    ELSE IF rotated_list[low] <= rotated_list[mid] THEN
      IF target >= rotated_list[low] AND target < rotated_list[mid] THEN
        high = mid - 1
      ELSE
        low = mid + 1
      END IF
    ELSE
      IF target > rotated_list[mid] AND target <= rotated_list[high] THEN
        low = mid + 1
      ELSE
        high = mid - 1
      END IF
    END IF
  END WHILE
  RETURN -1
END FUNCTION
