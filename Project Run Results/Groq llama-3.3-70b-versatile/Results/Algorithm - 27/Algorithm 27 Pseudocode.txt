# Rotate a square 2D matrix 90 degrees clockwise in-place
FUNCTION ROTATE_MATRIX(matrix)
  n = LENGTH(matrix)
  FOR layer FROM 0 TO n / 2
    first = layer
    last = n - layer - 1
    FOR i FROM first TO last
      top = matrix[layer][i]
      matrix[layer][i] = matrix[last - i + first][first]
      matrix[last - i + first][first] = matrix[last][last - i + first]
      matrix[last][last - i + first] = matrix[i][last]
      matrix[i][last] = top
    END FOR
  END FOR
  RETURN matrix
END FUNCTION
