# Integer Square Root (Binary Search)

FUNCTION intSqrt(n):
    IF n == 0 THEN RETURN 0
    low ← 1, high ← n
    WHILE low <= high:
        mid ← (low + high) DIV 2
        IF mid * mid == n THEN RETURN mid
        ELSE IF mid * mid < n THEN
            result ← mid
            low ← mid + 1
        ELSE high ← mid - 1
    RETURN result
