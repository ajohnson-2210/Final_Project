END FUNCTION 33. Square root using binary search (rounded
down)PlaintextFUNCTION IntegerSquareRoot(n): IF n < 2 THEN RETURN n

    low = 1
    high = n DIV 2
    ans = 0

    WHILE low <= high DO
        mid = low + (high - low) DIV 2
        IF mid * mid == n THEN
            RETURN mid
        ELSE IF mid * mid < n THEN
            ans = mid
            low = mid + 1
        ELSE
            high = mid - 1
        END IF
    END WHILE

    RETURN ans

41. Solve the N Queens problemPlaintextFUNCTION SolveNQueens(N): board =
    N x N grid initialized to 0 solutions = empty List

    PROCEDURE Backtrack(row, cols, diag1, diag2): IF row == N THEN
    APPEND Copy(board) TO solutions RETURN END IF

        FOR col FROM 0 TO N - 1 DO
            IF col IN cols OR (row - col) IN diag1 OR (row + col) IN diag2 THEN
                CONTINUE
            END IF

            board[row][col] = 1
            Backtrack(row + 1, cols + {col}, diag1 + {row - col}, diag2 + {row + col})
            board[row][col] = 0 // Backtrack
        END FOR

    END PROCEDURE

    Backtrack(0, empty Set, empty Set, empty Set) RETURN solutions END
    FUNCTION
