# Search in a rotated sorted array (Logarithmic time)    
FUNCTION SearchRotatedArray(arr, target):
    low = 0
    high = length(arr) - 1
    
    WHILE low <= high DO
        mid = low + (high - low) DIV 2
        IF arr[mid] == target THEN RETURN mid
        
        // Left side is sorted
        IF arr[low] <= arr[mid] THEN
            IF arr[low] <= target AND target < arr[mid] THEN
                high = mid - 1
            ELSE
                low = mid + 1
            END IF
        // Right side is sorted
        ELSE
            IF arr[mid] < target AND target <= arr[high] THEN
                low = mid + 1
            ELSE
                high = mid - 1
            END IF
        END IF
    END WHILE
    
    RETURN -1
END FUNCTION
