# Recursive search for file type in nested folders    
FUNCTION SearchFiles(directory, target_extension):
    matching_files = empty List
    
    FOR EACH item IN directory.getItems() DO
        IF item.isFolder() THEN
            matching_files.extend(SearchFiles(item, target_extension))
        ELSE IF item.getExtension() == target_extension THEN
            APPEND item.path TO matching_files
        END IF
    END FOR
    
    RETURN matching_files
END FUNCTION
