For a fixed window size sliding across an array return the maximum value within each window position


arr = [1, 3, -1, -3, 5, 3, 6, 7]
k = 2

results = []
start = 0
loop = True

while loop:
    end = start + k

    if end > len(arr):
        break
    
    index = start
    innerloop = True
    maxval = arr[start]

    while innerloop:
        try:
            if arr[index] > maxval:
                maxval = arr[index]
            index += 1
        except:
            innerloop = False

        if index >= end:
            innerloop = False

    results.append(maxval)
    start += 1

print(f"Sliding Window Maximums From Window Size {k}:")
print(results)
