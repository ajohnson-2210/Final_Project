Find the continuous subarray with the maximum sum (Kadane's algorithm)


import random

def kadane(arr):
    n = len(arr)

    current_sum = arr[0]
    best_sum = arr[0]

    current_start = 0
    best_ranges = [(0, 0)]

    i = 1
    while i < n:

        # Decide whether to extend or restart
        if current_sum + arr[i] < arr[i]:
            current_sum = arr[i]
            current_start = i
        else:
            current_sum = current_sum + arr[i]

        # New best sum then reset list
        if current_sum > best_sum:
            best_sum = current_sum
            best_ranges = [(current_start, i)]

        # Equal best sum then add another range
        elif current_sum == best_sum:
            best_ranges.append((current_start, i))

        i += 1

    return best_sum, best_ranges

# Generate random array (max 10 elements, mix of ints + floats)
size = random.randint(5, 10)

arr = []
i = 0
while i < size:
    if random.random() < 0.5:
        num = random.randint(-10, 10)      # integer
    else:
        num = round(random.uniform(-10, 10), 2)  # float
    arr.append(num)
    i += 1

print("Random array:", arr)

best_sum, ranges = kadane(arr)

print("\nMaximum subarray sum:", best_sum)
print("Number of max subarrays:", len(ranges))

idx = 0
while idx < len(ranges):
    start, end = ranges[idx]
    print(f"\nSubarray {idx+1} Start Index: {start}: ")
    j = start
    while j <= end:
        print(arr[j], end=", ")
        j += 1
    idx += 1
