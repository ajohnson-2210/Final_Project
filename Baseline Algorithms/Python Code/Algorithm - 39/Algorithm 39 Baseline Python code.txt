Detect a cycle in a single linked list (Floyd's Cycle Detection)

class Node:
    def __init__(self, value):
        self.value = value
        self.next = None


class LinkedList:
    def __init__(self):
        self.head = None
        self.tail = None
        self.nodes = []   # stores ALL nodes in order
        print("Linked List Created\n")

    # -----------------------------------------
    # APPEND
    # -----------------------------------------
    def append(self, value):
        new_node = Node(value)
        print(f"Appended Value {value}")
        if self.head is None:
            self.head = new_node
            self.tail = new_node
        else:
            self.tail.next = new_node
            self.tail = new_node

        self.nodes.append(new_node)

    # -----------------------------------------
    # PREPEND
    # -----------------------------------------
    def prepend(self, value):
        print(f"Prepended Value {value} To The Front")
        new_node = Node(value)
        if self.head is None:
            self.head = new_node
            self.tail = new_node
        else:
            new_node.next = self.head
            self.head = new_node

        self.nodes.insert(0, new_node)

    # -----------------------------------------
    # INSERT AT INDEX
    # -----------------------------------------
    def insert_at(self, index, value):
        print(f"Inserting Value {value} at Index {index}")

        if index == 0:
            self.prepend(value)
            return

        new_node = Node(value)
        temp = self.head
        i = 0

        while temp is not None and i < index - 1:
            temp = temp.next
            i += 1

        if temp is None:
            print("Insert Failed: Index out of range")
            return

        new_node.next = temp.next
        temp.next = new_node

        if new_node.next is None:
            self.tail = new_node

        self.nodes.insert(index, new_node)
        print(f"Insert Successful: {value} Inserted at Index {index}")

    # -----------------------------------------
    # CREATE CYCLE USING START + END INDEXES
    # -----------------------------------------
    def create_cycle(self, start_index, end_index):
        print(f"\nCycle Created: Loop Start index {start_index} → End index  {end_index}")
        if start_index < 0 or end_index < 0:
            print("Cycle Creation Failed: Invalid Index")
            return
        if start_index >= len(self.nodes) or end_index >= len(self.nodes):
            print("Cycle Creation Failed: Index Out Of Range")
            return

        start_node = self.nodes[start_index]
        end_node = self.nodes[end_index]

        end_node.next = start_node
        print(f"\nCycle Created: Node {start_index} now points to node {end_index}")

    # -----------------------------------------
    # REMOVE CYCLE
    # -----------------------------------------
    def remove_cycle(self):
        print("Remove Cycle Requested")

        slow = self.head
        fast = self.head

        # Detect cycle
        while fast and fast.next:
            slow = slow.next
            fast = fast.next.next
            if slow == fast:
                print("Cycle Detected: Proceeding with Removal")
                break
        else:
            print("No cycle found — nothing to remove")
            return

        # Find cycle start
        slow = self.head
        while slow != fast:
            slow = slow.next
            fast = fast.next

        cycle_start = slow

        # Find node just before cycle start
        temp = cycle_start
        while temp.next != cycle_start:
            temp = temp.next

        # Break the cycle
        temp.next = None

        # Update tail
        self.tail = self.nodes[-1]

        print(f"Cycle Removed Successfully: Loop started at Node with Value {cycle_start.value}")

    # -----------------------------------------
    # LOOP DIAGRAM
    # -----------------------------------------
    def loop_diagram(self):
        print("\nGenerating Loop Diagram...\n")

        nodes = self.nodes  # full list guaranteed

        # Detect cycle start + end
        visited = {}
        temp = self.head
        index = 0
        cycle_start = None
        cycle_end = None

        while temp is not None:
            if temp in visited:
                cycle_start = visited[temp]
                cycle_end = index - 1
                break
            visited[temp] = index
            temp = temp.next
            index += 1

        # Build chain text
        parts = [str(node.value) for node in nodes]
        sep = " → "
        chain = sep.join(parts)
        print(chain)

        if cycle_start is None:
            print("No Cycle Detected\n")
            return

        # Compute exact character positions
        start_char = sum(len(parts[i]) + len(sep) for i in range(cycle_start))
        start_center = start_char + len(parts[cycle_start]) // 2

        end_char = sum(len(parts[i]) + len(sep) for i in range(cycle_end))
        end_center = end_char + len(parts[cycle_end]) // 2

        # Bottom connector
        print(" " * start_center + "└" + "─" * (end_center - start_center - 1) + "┘")

        # Cycle message
        print(f"Cycle Detected From Node {cycle_start} to Node {cycle_end}\n")


# -----------------------------------------
# EXAMPLE USAGE
# -----------------------------------------
print("Detect Cycle in Singly Linked List\n")
l = LinkedList()
l.append(10)
l.append(20)
l.append(30)
l.append(40)
l.append(50)
l.append(60)


l.loop_diagram()

# Remove cycle
l.remove_cycle()

l.loop_diagram()

l.append(40)
l.append(50)

# Create cycle: 50 → 20
l.create_cycle(start_index=1, end_index=6)

l.loop_diagram()

# Remove cycle
l.remove_cycle()

print("\nAfter Removing Cycle:")
l.loop_diagram()
