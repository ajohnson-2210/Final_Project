Write a process that converts numerical scores into grades such as pass merit or distinction

loop = True
grade = ""
while loop == True:
    score = input("Enter Student Score: ")
    try:
        score = float(score)
        loop = False
    except:
        print(f"That's Not a Number, Please Try Again!")
    try:
        loop = True
        if int(score) < 0 or int(score) > 100:
            raise ValueError()
        else:
            loop = False
    except ValueError:
        print("Score out of range 0 - 100")
if score >= 60 and score < 70:
    print(f"Congratulations, You received a Grade of Pass!")
elif score >= 70 and score < 80:
    print(f"Congratulations, You received a Grade of Merit!")
elif score >= 80 and score < 100:
    print(f"Congratulations, You received a Grade of Distinction!")
elif score == 100:
    print(f"WOW!! you got a Perfect score Excellent!!, Congratulations, You received a Grade of Distinction!")
    
else:
    print(f"Unfortunately, You did Not pass this time!")
