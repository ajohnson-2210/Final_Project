Swap 2 variables without using a third variable

var1 <- 3
var2 <- STRING("tree")

OUTPUT STRING("variable 1 == {var1}")
OUTPUT STRING("variable 2 == {var2}")

IF CALL TYPE_OF(var1)== NUMBER AND CALL TYPE_OF(var2)== NUMBER THEN

    var1 <- var1 + var2
    var2 <- var1 - var2
    var1 <- var1 - var2

ELSE IF CALL TYPE_OF(var1)== STRING AND CALL TYPE_OF(var2)== STRING THEN

    var1 <- var1 + var2
    var2 <- CALL SUBSTRING(var1, 0, LENGTH OF var1 - LENGTH OF var2)
    var1 <- CALL SUBSTRING(var1, LENGTH OF var2, LENGTH OF var1)

ELSE

    var1 <- CALL TO_STRING(var1)

    var1 <- var1 + var2
    var2 <- CALL SUBSTRING(var1, 0, LENGTH OF var1 - LENGTH OF var2)
    var1 <- CALL SUBSTRING(var1, LENGTH OF var2, LENGTH OF var1)

ENDIF

OUTPUT STRING("variable 1 == {var1}")
OUTPUT STRING("variable 2 == {var2}")
