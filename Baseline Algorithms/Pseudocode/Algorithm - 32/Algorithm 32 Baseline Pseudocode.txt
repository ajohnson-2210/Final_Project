FUNCTION clean_text ACCEPTS(t)
    t <- REPLACE(t, STRING("’"), STRING("'"))
    t <- REPLACE(t, STRING("‘"), STRING("'"))
    t <- REPLACE(t, STRING("“"), STRING("))
    t <- REPLACE(t, STRING("”"), STRING("))
    t <- REPLACE(t, STRING("—"), STRING("-"))
    t <- REPLACE(t, STRING("–"), STRING("-"))
    t <- REPLACE(t, STRING("…"), STRING("..."))
    RETURN t
ENDFUNCTION

FUNCTION split_sentences ACCEPTS(text)
    temp <- REPLACE(text, STRING("?"), STRING("."))
    temp <- REPLACE(temp, STRING("!"), STRING("."))
    parts <- CALL SPLIT(temp, STRING("."))

    sentences <- EMPTY LIST
    i <- 0
    loop <- BOOLEAN(TRUE)

    WHILE loop == BOOLEAN(TRUE) DO
        TRY
            s <- TRIM(parts[i])
            IF LENGTH OF s > 0 THEN
                APPEND s TO sentences
            ENDIF
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE

    RETURN sentences
ENDFUNCTION

FUNCTION tokenise ACCEPTS(text)
    temp <- CALL LOWERCASE(text)
    temp <- REPLACE(temp, STRING("."), EMPTY STRING)
    temp <- REPLACE(temp, STRING(","), EMPTY STRING)
    temp <- REPLACE(temp, STRING(";"), EMPTY STRING)
    temp <- REPLACE(temp, STRING(":"), EMPTY STRING)
    RETURN CALL SPLIT(temp)
ENDFUNCTION

FUNCTION auto_generate_places ACCEPTS()
    endings <- EMPTY LIST
    i <- 0
    loop <- BOOLEAN(TRUE)
    WHILE loop == BOOLEAN(TRUE) DO
        TRY
            w <- all_words[i]
            IF w ENDSWITH ANY OF [
                CALL TO_STRING(valley), CALL TO_STRING(land), CALL TO_STRING(road),
                CALL TO_STRING(house), CALL TO_STRING(place), CALL TO_STRING(meadow), CALL TO_STRING(bank)
            ] THEN
                j <- 0
                found <- BOOLEAN(FALSE)
                WHILE j < LENGTH OF w AND found == BOOLEAN(FALSE) DO
                    TRY
                        suffix <- CALL SUBSTRING(w, j, LENGTH OF w)
                        IF suffix IN LIST [
                            CALL TO_STRING(valley), CALL TO_STRING(land), CALL TO_STRING(road),
                            CALL TO_STRING(house), CALL TO_STRING(place), CALL TO_STRING(meadow), CALL TO_STRING(bank)
                        ] THEN
                            APPEND suffix TO endings
                            found <- BOOLEAN(TRUE)
                            BREAK
                        ENDIF
                        j <- j + 1
                    CATCH
                        j <- LENGTH OF w
                    ENDTRY
                ENDWHILE
            ENDIF
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE

    places <- EMPTY LIST
    i <- 0
    loop <- BOOLEAN(TRUE)
    WHILE loop == BOOLEAN(TRUE) DO
        TRY
            w <- all_words[i]
            j <- 0
            j_loop <- BOOLEAN(TRUE)
            WHILE j_loop == BOOLEAN(TRUE) DO
                TRY
                    IF w ENDSWITH endings[j] THEN
                        APPEND w TO places
                    ENDIF
                    j <- j + 1
                CATCH
                    j_loop <- BOOLEAN(FALSE)
                ENDTRY
            ENDWHILE
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE
    RETURN places
ENDFUNCTION

FUNCTION auto_generate_animals ACCEPTS()
    animals <- EMPTY LIST

    raw_words_clean <- raw_text
    raw_words_clean <- REPLACE(raw_words_clean, STRING(","), STRING(","))
    raw_words_clean <- REPLACE(raw_words_clean, STRING("."), STRING("."))
    raw_words_clean <- REPLACE(raw_words_clean, STRING(";"), STRING(";"))
    raw_words_clean <- REPLACE(raw_words_clean, STRING(":"), STRING(":"))
    raw_words_clean <- REPLACE(raw_words_clean, STRING("!"), STRING("!"))
    raw_words_clean <- REPLACE(raw_words_clean, STRING("?"), STRING("?"))

    raw_words_list <- CALL SPLIT(raw_words_clean)

    FUNCTION looks_like_animal ACCEPTS(w)
        w <- CALL LOWERCASE(w)
        IF w ENDSWITH CALL TO_STRING(mole) THEN RETURN BOOLEAN(TRUE)
        IF w ENDSWITH CALL TO_STRING(rat) THEN RETURN BOOLEAN(TRUE)
        IF w ENDSWITH CALL TO_STRING(otter) THEN RETURN BOOLEAN(TRUE)
        IF w ENDSWITH CALL TO_STRING(badger) THEN RETURN BOOLEAN(TRUE)
        IF w == CALL TO_STRING(animal) THEN RETURN BOOLEAN(TRUE)
        RETURN BOOLEAN(FALSE)
    ENDFUNCTION

    i <- 0
    loop <- BOOLEAN(TRUE)
    WHILE loop == BOOLEAN(TRUE) DO
        TRY
            IF raw_words_list[i] == CALL TO_STRING(was) AND i > 0 AND raw_words_list[i - 1] == CALL TO_STRING(it) THEN
                nxt <- CALL LOWERCASE(raw_words_list[i + 1])
                IF looks_like_animal(nxt) THEN
                    APPEND nxt TO animals
                ENDIF
            ENDIF
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE

    raw_words <- CALL SPLIT(REPLACE(REPLACE(raw_text, STRING(";"), STRING("")), STRING(":"), STRING("")))

    i <- 0
    loop <- BOOLEAN(TRUE)
    WHILE loop == BOOLEAN(TRUE) DO
        TRY
            w <- raw_words[i]
            IF FIRSTCHAR(w) IS UPPERCASE THEN
                lw <- CALL LOWERCASE(w)
                lw <- REPLACE(lw, STRING(","), EMPTY STRING)
                lw <- REPLACE(lw, STRING("."), EMPTY STRING)
                lw <- REPLACE(lw, STRING("!"), EMPTY STRING)
                lw <- REPLACE(lw, STRING("?"), EMPTY STRING)
                IF looks_like_animal(lw) THEN
                    APPEND lw TO animals
                ENDIF
            ENDIF
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE

    i <- 0
    loop <- BOOLEAN(TRUE)
    WHILE loop == BOOLEAN(TRUE) DO
        TRY
            w <- all_words[i]
            IF looks_like_animal(w) THEN
                APPEND w TO animals
            ENDIF
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE

    unique <- EMPTY LIST
    i <- 0
    loop <- BOOLEAN(TRUE)
    WHILE loop == BOOLEAN(TRUE) DO
        TRY
            IF animals[i] NOT IN unique THEN
                APPEND animals[i] TO unique
            ENDIF
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE

    RETURN unique
ENDFUNCTION

FUNCTION words_match ACCEPTS(w1, w2)
    IF w1 == w2 THEN RETURN BOOLEAN(TRUE)

    i <- 0
    loop <- BOOLEAN(TRUE)

    WHILE loop == BOOLEAN(TRUE) DO
        TRY
            a, b <- synonyms[i]
            IF (w1 == a AND w2 == b) OR (w1 == b AND w2 == a) THEN
                RETURN BOOLEAN(TRUE)
            ENDIF
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE

    RETURN BOOLEAN(FALSE)
ENDFUNCTION

                        score <- score + 3
                    ENDIF
                ENDIF


                IF LENGTH OF q_words > 0 AND
                   (q_words[0] == CALL TO_STRING(did) OR q_words[0] == CALL TO_STRING(does) OR q_words[0] == CALL TO_STRING(do)) THEN

                    subject <- EMPTY STRING
                    i2 <- 1
                    loop2 <- BOOLEAN(TRUE)

                    WHILE loop2 == BOOLEAN(TRUE) DO
                        TRY
                            IF q_words[i2] NOT IN stopwords THEN
                                subject <- q_words[i2]
                                loop2 <- BOOLEAN(FALSE)
                            ENDIF
                            i2 <- i2 + 1
                        CATCH
                            loop2 <- BOOLEAN(FALSE)
                        ENDTRY
                    ENDWHILE

                    obj <- EMPTY STRING
                    i2 <- 0
                    loop2 <- BOOLEAN(TRUE)

                    WHILE loop2 == BOOLEAN(TRUE) DO
                        TRY
                            IF q_words[i2] NOT IN stopwords THEN
                                obj <- q_words[i2]
                            ENDIF
                            i2 <- i2 + 1
                        CATCH
                            loop2 <- BOOLEAN(FALSE)
                        ENDTRY
                    ENDWHILE

                    s2 <- 0
                    loop2 <- BOOLEAN(TRUE)

                    WHILE loop2 == BOOLEAN(TRUE) DO
                        TRY
                            IF s_words[s2] == subject THEN
                                score <- score + 3
                            ENDIF
                            IF s_words[s2] == obj THEN
                                score <- score + 4
                            ENDIF
                            s2 <- s2 + 1
                        CATCH
                            loop2 <- BOOLEAN(FALSE)
                        ENDTRY
                    ENDWHILE

                    s2 <- 0
                    loop2 <- BOOLEAN(TRUE)

                    WHILE loop2 == BOOLEAN(TRUE) DO
                        TRY
                            IF s_words[s2] IN LIST [
                                CALL TO_STRING(was), CALL TO_STRING(were), CALL TO_STRING(had),
                                CALL TO_STRING(approached), CALL TO_STRING(contained), CALL TO_STRING(included)
                            ] THEN
                                score <- score + 2
                            ENDIF
                            s2 <- s2 + 1
                        CATCH
                            loop2 <- BOOLEAN(FALSE)
                        ENDTRY
                    ENDWHILE
                ENDIF

                s <- s + 1
            CATCH
                s_loop <- BOOLEAN(FALSE)
            ENDTRY
        ENDWHILE


        IF i > 0 THEN
            prev_words <- tokenise(sentences[i - 1])

            q <- 0
            q_loop <- BOOLEAN(TRUE)

            WHILE q_loop == BOOLEAN(TRUE) DO
                TRY
                    qw <- q_words[q]
                    IF qw NOT IN stopwords THEN
                        p <- 0
                        p_loop <- BOOLEAN(TRUE)

                        WHILE p_loop == BOOLEAN(TRUE) DO
                            TRY
                                IF words_match(prev_words[p], qw) THEN
                                    score <- score + 1
                                ENDIF
                                p <- p + 1
                            CATCH

                            ENDTRY
                        ENDWHILE
                    ENDIF
                    q <- q + 1
                CATCH
                    q_loop <- BOOLEAN(FALSE)
                ENDTRY
            ENDWHILE
        ENDIF

            next_words <- tokenise(sentences[i + 1])

                        ENDWHILE
                    ENDIF
                    q <- q + 1
                CATCH
                    q_loop <- BOOLEAN(FALSE)
                ENDTRY
            ENDWHILE
        ENDIF


        APPEND (score, sent) TO scores
        i <- i + 1
    CATCH
        loop <- BOOLEAN(FALSE)
    ENDTRY
ENDWHILE

FUNCTION rewrite_answer ACCEPTS(sentence, question_type)
    words <- tokenise(sentence)

    IF LENGTH OF words > 0 THEN
        IF words[0] == CALL TO_STRING(because) THEN
            words <- CALL SUBLIST(words, 1, LENGTH OF words)
        ENDIF
    ENDIF

    IF question_type == CALL TO_STRING(where) THEN
        i <- 0
        loop <- BOOLEAN(TRUE)
        WHILE loop == BOOLEAN(TRUE) DO
            TRY
                IF words[i] == CALL TO_STRING(at) THEN
                    words[i] <- CALL TO_STRING(in)
                ENDIF
                i <- i + 1
            CATCH
                loop <- BOOLEAN(FALSE)
            ENDTRY
        ENDWHILE
    ENDIF

    IF question_type == CALL TO_STRING(why) THEN
        i <- 0
        loop <- BOOLEAN(TRUE)
        start <- 0
        WHILE loop == BOOLEAN(TRUE) DO
            TRY
                IF words[i] == CALL TO_STRING(because) THEN
                    start <- i
                ENDIF
                i <- i + 1
            CATCH
                loop <- BOOLEAN(FALSE)
            ENDTRY
        ENDWHILE
        words <- CALL SUBLIST(words, start, LENGTH OF words)
    ENDIF

    IF question_type == CALL TO_STRING(who) THEN
        IF LENGTH OF words > 8 THEN
            new_words <- EMPTY LIST
            j <- 0
            loop2 <- BOOLEAN(TRUE)
            WHILE loop2 == BOOLEAN(TRUE) DO
                TRY
                    IF j < 8 THEN
                        APPEND words[j] TO new_words
                    ENDIF
                    j <- j + 1
                CATCH
                    loop2 <- BOOLEAN(FALSE)
                ENDTRY
            ENDWHILE
            words <- new_words
        ENDIF
    ENDIF

    IF question_type == CALL TO_STRING(what) THEN
        i <- 0
        loop <- BOOLEAN(TRUE)
        start <- 0
        found <- BOOLEAN(FALSE)
        WHILE loop == BOOLEAN(TRUE) DO
            TRY
                IF words[i] == CALL TO_STRING(for) THEN
                    start <- i
                    found <- BOOLEAN(TRUE)
                ENDIF
                i <- i + 1
            CATCH
                loop <- BOOLEAN(FALSE)
            ENDTRY
        ENDWHILE
        IF found == BOOLEAN(TRUE) THEN
            new_words <- EMPTY LIST
            j <- start
            loop2 <- BOOLEAN(TRUE)
            WHILE loop2 == BOOLEAN(TRUE) DO
                TRY
                    APPEND words[j] TO new_words
                    j <- j + 1
                CATCH
                    loop2 <- BOOLEAN(FALSE)
                ENDTRY
            ENDWHILE
            words <- new_words
        ENDIF
    ENDIF

    rewritten <- EMPTY STRING
    i <- 0
    loop <- BOOLEAN(TRUE)

    WHILE loop == BOOLEAN(TRUE) DO
        TRY
            rewritten <- rewritten + words[i] + STRING("")
            i <- i + 1
        CATCH
            loop <- BOOLEAN(FALSE)
        ENDTRY
    ENDWHILE

    RETURN TRIM(rewritten)
ENDFUNCTION
