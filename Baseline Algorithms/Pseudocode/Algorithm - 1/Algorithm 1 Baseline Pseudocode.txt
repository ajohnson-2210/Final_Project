Determine if 2 strings are anagrams of each other

string <- STRING("1a23")
string1 <- EMPTY STRING
FOR EACH char IN string DO
    IF CALL ISALPHA(char) THEN
        string1 <- string1 + CALL LOWERCASE(char)
    ENDIF
ENDFOR

string <- STRING("45a6")
string2 <- EMPTY STRING
FOR EACH char IN string DO
    IF CALL ISALPHA(char) THEN
        string2 <- string2 + CALL LOWERCASE(char)
    ENDIF
ENDFOR

list1 <- CALL CHARACTERS_OF(string1)
list2 <- CALL CHARACTERS_OF(string2)

IF (LENGTH OF list1 == 0) OR (LENGTH OF list2 == 0) OR (LENGTH OF list1 != LENGTH OF list2) THEN
    OUTPUT STRING("string1 and string2 are Not anagrams of each other")
    RETURN
ENDIF

list1 <- CALL SORT(list1)
list2 <- CALL SORT(list2)

WHILE LENGTH OF list1 > 0 DO
    x <- CALL REMOVE_LAST(list1)
    y <- CALL REMOVE_LAST(list2)

    IF x != y THEN
        OUTPUT STRING("string1 and string2 are Not anagrams of each other")
        RETURN
    ENDIF
ENDWHILE
