Implement a Stack using an Array (push/pop/peek)

CLASS Stack
    CONSTRUCTOR ACCEPTS()
        COMMENT PROPERTY GROUP(
            THIS.items <- EMPTY LIST,
            THIS.top <- -1
        )
    ENDCONSTRUCTOR

    FUNCTION show ACCEPTS()
        OUTPUT STRING("Stack: {THIS.items}")
    ENDFUNCTION

    FUNCTION push ACCEPTS(value)
        new_length <- LENGTH OF THIS.items + 1
        temp <- LIST OF SIZE EQUALS new_length FILLED WITH CALL NULL()

        i <- 0
        loop <- BOOLEAN(TRUE)

        WHILE loop == BOOLEAN(TRUE) DO
            TRY
                temp[i] <- THIS.items[i]
                i <- i + 1
            CATCH
                loop <- BOOLEAN(FALSE)
            ENDTRY
        ENDWHILE

        temp[THIS.top + 1] <- value
        THIS.items <- temp
        THIS.top <- THIS.top + 1

        OUTPUT STRING("Pushed: {value}")
        THIS.show()
    ENDFUNCTION	

    FUNCTION pop ACCEPTS()
        IF THIS.is_empty() == BOOLEAN(TRUE) THEN
            OUTPUT STRING("Pop attempted on empty stack")
            THIS.show()
            RETURN CALL NULL()
        ENDIF

        value <- THIS.items[THIS.top]

        new_length <- LENGTH OF THIS.items - 1
        temp <- LIST OF SIZE EQUALS new_length FILLED WITH CALL NULL()

        i <- 0
        WHILE i < new_length DO
            temp[i] <- THIS.items[i]
            i <- i + 1
        ENDWHILE

        THIS.items <- temp
        THIS.top <- THIS.top - 1

        OUTPUT STRING("Popped: {value}")
        THIS.show()
        RETURN value
    ENDFUNCTION


    FUNCTION peek ACCEPTS()
        IF THIS.is_empty() == BOOLEAN(TRUE) THEN
            OUTPUT STRING("Peek attempted on empty stack")
            THIS.show()
            RETURN CALL NULL()
        ENDIF

        value <- THIS.items[THIS.top]
        OUTPUT STRING("Peeked: {value}")
        THIS.show()
        RETURN value
    ENDFUNCTION


    FUNCTION is_empty ACCEPTS()
        RETURN THIS.top == -1
    ENDFUNCTION


    FUNCTION size ACCEPTS()
        RETURN THIS.top + 1
    ENDFUNCTION

ENDCLASS


s <- CALL Stack()

s.push(10)
s.push(20)
s.push(30)

s.peek()
s.pop()
s.peek()
s.pop()
s.pop()
s.pop()
s.peek()
