Find all pairs in an array that match a given number

COMMENT External library: random
FUNCTION find_pairs ACCEPTS(arr, target, decimals)
    pairs <- EMPTY LIST
    i <- 0
    n <- LENGTH OF arr

    WHILE i < n DO
        j <- i + 1

        WHILE j < n DO
            IF CALL ROUND(arr[i] + arr[j], decimals) == CALL ROUND(target, decimals) THEN
                APPEND TUPLE(arr[i], arr[j]) TO pairs
            ENDIF

            j <- j + 1
        ENDWHILE

        i <- i + 1
    ENDWHILE

    RETURN pairs
ENDFUNCTION



# ============================================================
# MAIN PROGRAM FLOW — PAIR FINDER
# ============================================================

size <- 20
arr <- EMPTY LIST
i <- 0

WHILE i < size DO
    rand_val <- RANDOM_FLOAT(0, 1)

    IF rand_val < 0.5 THEN
        num <- RANDOM_INT(-20, 20)
    ELSE
        num <- CALL ROUND(RANDOM_FLOAT(-20, 20), 2)
    ENDIF

    APPEND num TO arr
    i <- i + 1
ENDWHILE

OUTPUT STRING("Random array:")
OUTPUT STRING("{arr}")

target <- INPUT(STRING("Enter target number:"))

pairs <- find_pairs(arr, target, 6)

OUTPUT STRING("\\nPairs that sum to {target}:\\n")

i <- 0

IF LENGTH OF pairs == 0 THEN
    OUTPUT STRING("No matching pairs found.")
ELSE
    WHILE i < LENGTH OF pairs DO
        OUTPUT STRING("{pairs[i]}")
        i <- i + 1
    ENDWHILE
ENDIF
