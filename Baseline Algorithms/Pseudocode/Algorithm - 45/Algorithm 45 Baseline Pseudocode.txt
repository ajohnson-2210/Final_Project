Create instructions for adding 2 matrices of equal size

COMMENT External library: random

OUTPUT STRING("Steps to Add 2 Equally Sized Matrices.")

# ============================
# STEP 1: Choose matrix size
# ============================

rows <- RANDOM_INT(2, 5)
cols <- RANDOM_INT(2, 5)

OUTPUT STRING("STEP 1: Matrix size chosen: {rows} rows x {cols} columns\\n")

# ============================
# STEP 2: Generate Matrix A
# ============================

OUTPUT STRING("STEP 2: Generating Matrix A...\\n")

A <- EMPTY LIST
r <- 0

WHILE r < rows DO
    row_list <- EMPTY LIST
    c <- 0

    WHILE c < cols DO
        rand_val <- RANDOM_FLOAT(0, 1)

        IF rand_val < 0.5 THEN
            num <- RANDOM_INT(-10, 10)
        ELSE
            num <- CALL ROUND(RANDOM_FLOAT(-10, 10), 2)
        ENDIF

        APPEND num TO row_list
        c <- c + 1
    ENDWHILE

    APPEND row_list TO A
    r <- r + 1
ENDWHILE

OUTPUT STRING("Matrix A:\\n")

r <- 0
WHILE r < rows DO
    OUTPUT STRING("{A[r]}")
    r <- r + 1
ENDWHILE

OUTPUT STRING("\\n")


# ============================
# STEP 3: Generate Matrix B
# ============================

OUTPUT STRING("STEP 3: Generating Matrix B...\\n")

B <- EMPTY LIST
r <- 0

WHILE r < rows DO
    row_list <- EMPTY LIST
    c <- 0

    WHILE c < cols DO
        rand_val <- RANDOM_FLOAT(0, 1)

        IF rand_val < 0.5 THEN
            num <- RANDOM_INT(-10, 10)
        ELSE
            num <- CALL ROUND(RANDOM_FLOAT(-10, 10), 2)
        ENDIF

        APPEND num TO row_list
        c <- c + 1
    ENDWHILE

    APPEND row_list TO B
    r <- r + 1
ENDWHILE

OUTPUT STRING("Matrix B:\\n")

r <- 0
WHILE r < rows DO
    OUTPUT STRING("{B[r]}")
    r <- r + 1
ENDWHILE

OUTPUT STRING("\\n")


# ============================
# STEP 4: Add matrices (rounded 2dp)
# ============================

OUTPUT STRING("STEP 4: Adding Matrix A and Matrix B...\\n")

Result <- EMPTY LIST
r <- 0

OUTPUT STRING("Adding..")

WHILE r < rows DO
    row_list <- EMPTY LIST
    c <- 0

    WHILE c < cols DO
        sum_value <- A[r][c] + B[r][c]

        rounded_value <- (INTEGER(sum_value * 100)) / 100.0

        APPEND rounded_value TO row_list

        OUTPUT STRING("A[{r}][{c}] + B[{r}][{c}] → {A[r][c]} + {B[r][c]} == {rounded_value}")

        c <- c + 1
    ENDWHILE

    APPEND row_list TO Result
    r <- r + 1
ENDWHILE


# ============================
# STEP 5: Print all matrices
# ============================

OUTPUT STRING("\\nSTEP 5: Final Matrix\\n")

OUTPUT STRING("\\nA + B =")

r <- 0
WHILE r < rows DO
    OUTPUT STRING("{Result[r]}")
    r <- r + 1
ENDWHILE
