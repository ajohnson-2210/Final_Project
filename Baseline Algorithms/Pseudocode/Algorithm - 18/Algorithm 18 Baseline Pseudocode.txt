Compute the frequency of each character in a string

string1 <- STRING("This is a Sentence, that needs the chars counted!!")
stringlist <- CALL CHARACTERS_OF(string1)
countlist <- EMPTY DICTIONARY

WHILE LENGTH OF stringlist > 0 DO
    char <- CALL REMOVE_FIRST(stringlist)

    IF char NOT IN countlist THEN
        countlist[char] <- 1
    ELSE
        countlist[char] <- countlist[char] + 1
    ENDIF
ENDWHILE


FOR EACH entry IN CALL ITEMS(countlist) DO
    OUTPUT STRING("Character {key} appears {val} times")
ENDFOR
