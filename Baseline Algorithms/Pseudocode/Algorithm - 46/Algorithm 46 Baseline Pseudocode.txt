Develop a process for multiplying 2 matrices

COMMENT External library: random

OUTPUT STRING("Steps to Multiply 2 Matrices.")

# ============================
# STEP 1: Choose matrix sizes
# ============================

rowsA <- RANDOM INT(2, 4)
colsA <- RANDOM INT(2, 4)
colsB <- RANDOM INT(2, 4)

OUTPUT STRING("STEP 1: Matrix sizes chosen:")
OUTPUT STRING("Matrix A: {rowsA} x {colsA}")
OUTPUT STRING("Matrix B: {colsA} x {colsB}\\n")


# ============================
# STEP 2: Generate Matrix A
# ============================

OUTPUT STRING("STEP 2: Generating Matrix A...\\n")

A <- EMPTY LIST
r <- 0

WHILE r < rowsA DO
    row_list <- EMPTY LIST
    c <- 0

    WHILE c < colsA DO
        rand_val <- RANDOM FLOAT(0, 1)

        IF rand_val < 0.5 THEN
            num <- RANDOM INT(-10, 10)
        ELSE
            num <- CALL ROUND(RANDOM FLOAT(-10, 10), 2)
        ENDIF

        APPEND num TO row_list
        c <- c + 1
    ENDWHILE

    APPEND row_list TO A
    r <- r + 1
ENDWHILE

OUTPUT STRING("Matrix A:\\n")

r <- 0
WHILE r < rowsA DO
    OUTPUT STRING("{A[r]}")
    r <- r + 1
ENDWHILE

OUTPUT STRING("\\n")


# ============================
# STEP 3: Generate Matrix B
# ============================

OUTPUT STRING("STEP 3: Generating Matrix B...\\n")

B <- EMPTY LIST
r <- 0

WHILE r < colsA DO
    row_list <- EMPTY LIST
    c <- 0

    WHILE c < colsB DO
        rand_val <- RANDOM FLOAT(0, 1)

        IF rand_val < 0.5 THEN
            num <- RANDOM INT(-10, 10)
        ELSE
            num <- CALL ROUND(RANDOM FLOAT(-10, 10), 2)
        ENDIF

        APPEND num TO row_list
        c <- c + 1
    ENDWHILE

    APPEND row_list TO B
    r <- r + 1
ENDWHILE

OUTPUT STRING("Matrix B:\\n")

r <- 0
WHILE r < colsA DO
    OUTPUT STRING("{B[r]}")
    r <- r + 1
ENDWHILE

OUTPUT STRING("\\n")


# ============================
# STEP 4: Multiply matrices
# ============================

OUTPUT STRING("STEP 4: Multiplying Matrix A and Matrix B...\\n")
OUTPUT STRING("Multiplying...\\n")

Result <- EMPTY LIST
r <- 0

WHILE r < rowsA DO
    row_list <- EMPTY LIST
    c <- 0

    WHILE c < colsB DO
        k <- 0
        running_sum <- 0

        OUTPUT STRING("Element Result[{r}][{c}]:")

        WHILE k < colsA DO
            product <- A[r][k] * B[k][c]
            rounded_product <- (INTEGER(product * 100)) / 100.0
            running_sum <- running_sum + rounded_product

            OUTPUT STRING("A[{r}][{k}] * B[{k}][{c}] → {A[r][k]} * {B[k][c]} == {rounded_product}")

            k <- k + 1
        ENDWHILE

        final_value <- (INTEGER(running_sum * 100)) / 100.0
        APPEND final_value TO row_list

        OUTPUT STRING("Sum for Result[{r}][{c}] == {final_value}\\n")

        c <- c + 1
    ENDWHILE

    APPEND row_list TO Result
    r <- r + 1
ENDWHILE


# ============================
# STEP 5: Print final matrix
# ============================

OUTPUT STRING("\\nSTEP 5: Final Matrix\\n")
OUTPUT STRING("A × B =")

r <- 0
WHILE r < rowsA DO
    OUTPUT STRING("{Result[r]}")
    r <- r + 1
ENDWHILE
