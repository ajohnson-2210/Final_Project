Implement a queue using 2 stacks

CLASS Stack
    CONSTRUCTOR ACCEPTS(name)
        COMMENT PROPERTY GROUP(
            THIS.items <- EMPTY LIST,
            THIS.name <- name
        )
    ENDCONSTRUCTOR


    FUNCTION push ACCEPTS(value)
        APPEND value TO THIS.items
        OUTPUT STRING("{THIS.name}: pushed {value}")
    ENDFUNCTION


    FUNCTION pop ACCEPTS()
        IF LENGTH OF THIS.items == 0 THEN
            OUTPUT STRING("{THIS.name}: pop attempted on empty stack")
            RETURN CALL NULL()
        ENDIF

        value <- REMOVE LAST ITEM FROM THIS.items
        OUTPUT STRING("{THIS.name}: popped {value}")
        RETURN value
    ENDFUNCTION


    FUNCTION is_empty ACCEPTS()
        RETURN LENGTH OF THIS.items == 0
    ENDFUNCTION


    FUNCTION peek ACCEPTS()
        IF LENGTH OF THIS.items == 0 THEN
            RETURN CALL NULL()
        ENDIF

        RETURN THIS.items[LENGTH OF THIS.items - 1]
    ENDFUNCTION


    FUNCTION display ACCEPTS()
        OUTPUT STRING("{THIS.name} {THIS.items}")
    ENDFUNCTION

ENDCLASS



CLASS QueueUsingTwoStacks

    CONSTRUCTOR ACCEPTS()
        COMMENT PROPERTY GROUP(
            THIS.in_stack <- CALL Stack(STRING("(Stack1)IN")),
            THIS.out_stack <- CALL Stack(STRING("(Stack2)OUT"))
        )
        OUTPUT STRING("Queue created using two stacks\\n")
    ENDCONSTRUCTOR


    FUNCTION enqueue ACCEPTS(value)
        OUTPUT STRING("\\nEnqueue: {value}")
        THIS.in_stack.push(value)
        THIS.display()
    ENDFUNCTION


    FUNCTION dequeue ACCEPTS()
        OUTPUT STRING("\\nDequeue requested")

        IF THIS.out_stack.is_empty() == BOOLEAN(TRUE) THEN
            OUTPUT STRING("OUT is empty -> transferring from IN")

            WHILE THIS.in_stack.is_empty() == BOOLEAN(FALSE) DO
                moved <- THIS.in_stack.pop()
                THIS.out_stack.push(moved)
            ENDWHILE
        ENDIF

        IF THIS.out_stack.is_empty() == BOOLEAN(TRUE) THEN
            OUTPUT STRING("Queue is empty -- nothing to dequeue")
            RETURN CALL NULL()
        ENDIF

        value <- THIS.out_stack.pop()
        OUTPUT STRING("Dequeued: {value}")
        THIS.display()
        RETURN value
    ENDFUNCTION


    FUNCTION peek ACCEPTS()
        OUTPUT STRING("\\nPeek requested")

        IF THIS.out_stack.is_empty() == BOOLEAN(TRUE) THEN
            WHILE THIS.in_stack.is_empty() == BOOLEAN(FALSE) DO
                moved <- THIS.in_stack.pop()
                THIS.out_stack.push(moved)
            ENDWHILE
        ENDIF

        value <- THIS.out_stack.peek()
        OUTPUT STRING("Front of queue: {value}")
        THIS.display()
        RETURN value
    ENDFUNCTION


    FUNCTION display ACCEPTS()
        OUTPUT STRING("\\nQueue State:")
        THIS.in_stack.display()
        THIS.out_stack.display()
        OUTPUT STRING("")
    ENDFUNCTION

ENDCLASS



q <- CALL QueueUsingTwoStacks()

q.enqueue(10)
q.enqueue(20)
q.enqueue(30)

q.dequeue()

q.enqueue(40)
q.enqueue(50)

q.peek()

q.dequeue()
q.dequeue()
q.dequeue()
