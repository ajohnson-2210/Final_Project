Find the continuous subarray with the maximum sum (Kadane's algorithm)

COMMENT External library: random
FUNCTION kadane ACCEPTS(arr)
    n <- LENGTH OF arr
    current_sum <- arr[0]
    best_sum <- arr[0]
    current_start <- 0
    best_ranges <- LIST [TUPLE(0, 0)]
    i <- 1

    WHILE i < n DO
        IF current_sum + arr[i] < arr[i] THEN
            current_sum <- arr[i]
            current_start <- i
        ELSE
            current_sum <- current_sum + arr[i]
        ENDIF

        IF current_sum > best_sum THEN
            best_sum <- current_sum
            best_ranges <- LIST [TUPLE(current_start, i)]

        ELIF current_sum == best_sum THEN
            APPEND TUPLE(current_start, i) TO best_ranges
        ENDIF

        i <- i + 1
    ENDWHILE

    RETURN TUPLE(best_sum, best_ranges)
ENDFUNCTION



# ============================================================
# MAIN PROGRAM FLOW — KADANE
# ============================================================

size <- RANDOM_INT(5, 10)
arr <- EMPTY LIST
i <- 0

WHILE i < size DO
    rand_val <- RANDOM_FLOAT(0, 1)

    IF rand_val < 0.5 THEN
        num <- RANDOM_INT(-10, 10)
    ELSE
        num <- CALL ROUND(RANDOM_FLOAT(-10, 10), 2)
    ENDIF

    APPEND num TO arr
    i <- i + 1
ENDWHILE

OUTPUT STRING("Random array: {arr}")

result <- kadane(arr)
best_sum <- result[0]
ranges <- result[1]

OUTPUT STRING("\\nMaximum subarray sum: {best_sum}")
OUTPUT STRING("Number of max subarrays: {LENGTH OF ranges}")

idx <- 0

WHILE idx < LENGTH OF ranges DO
    start <- ranges[idx][0]
    end <- ranges[idx][1]

    OUTPUT STRING("\\nSubarray {idx + 1} Start Index: {start}:")

    j <- start
    WHILE j <= end DO
        OUTPUT STRING("{arr[j]},")
        j <- j + 1
    ENDWHILE

    idx <- idx + 1
ENDWHILE
