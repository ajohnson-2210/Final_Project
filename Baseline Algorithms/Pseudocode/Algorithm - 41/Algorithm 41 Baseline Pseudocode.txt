Solve the N Queens Problem for an arbitrary board size N

FUNCTION safe_placement ACCEPTS(board, row, col, n)
    r <- 0
    WHILE r < row DO
        IF board[r] == col THEN
            RETURN BOOLEAN(FALSE)
        ENDIF
        r <- r + 1
    ENDWHILE

    r <- row - 1
    c <- col - 1
    WHILE r >= 0 AND c >= 0 DO
        IF board[r] == c THEN
            RETURN BOOLEAN(FALSE)
        ENDIF
        r <- r - 1
        c <- c - 1
    ENDWHILE

    r <- row - 1
    c <- col + 1
    WHILE r >= 0 AND c < n DO
        IF board[r] == c THEN
            RETURN BOOLEAN(FALSE)
        ENDIF
        r <- r - 1
        c <- c + 1
    ENDWHILE

    RETURN BOOLEAN(TRUE)
ENDFUNCTION



FUNCTION solve_n_queens_util ACCEPTS(board, row, n, solutions)
    IF row == n THEN
        APPEND board[:] TO solutions
        RETURN
    ENDIF

    col <- 0
    WHILE col < n DO
        IF safe_placement(board, row, col, n) == BOOLEAN(TRUE) THEN
            board[row] <- col
            solve_n_queens_util(board, row + 1, n, solutions)
            board[row] <- -1
        ENDIF
        col <- col + 1
    ENDWHILE
ENDFUNCTION



FUNCTION solve_n_queens ACCEPTS(n)
    board <- LIST OF SIZE EQUALS n FILLED WITH -1
    solutions <- EMPTY LIST
    solve_n_queens_util(board, 0, n, solutions)
    RETURN solutions
ENDFUNCTION



FUNCTION print_board ACCEPTS(solution)
    n <- LENGTH OF solution
    r <- 0

    WHILE r < n DO
        row_str <- STRING("")
        c <- 0

        WHILE c < n DO
            IF solution[r] == c THEN
                row_str <- row_str + CALL TO_STRING(Q)
            ELSE
                row_str <- row_str + STRING(".")
            ENDIF
            c <- c + 1
        ENDWHILE

        OUTPUT STRING("{row_str}")
        r <- r + 1
    ENDWHILE

    OUTPUT STRING("")
ENDFUNCTION



FUNCTION print_all_solutions ACCEPTS(solutions)
    total <- LENGTH OF solutions
    OUTPUT STRING("\\nShowing ALL {total} solutions:\\n")

    i <- 0
    WHILE i < total DO
        OUTPUT STRING("Solution {i + 1}:")
        print_board(solutions[i])
        i <- i + 1
    ENDWHILE
ENDFUNCTION



# ============================================================
# MAIN PROGRAM FLOW — N QUEENS
# ============================================================

OUTPUT STRING("Enter board size N:")
N <- INPUT(STRING("Enter board size N"))

solutions <- solve_n_queens(N)
total <- LENGTH OF solutions

OUTPUT STRING("\\nSolved {N}-Queens.")
OUTPUT STRING("Total solutions: {total}\\n")

index <- 0
loop <- BOOLEAN(TRUE)

WHILE loop == BOOLEAN(TRUE) DO
    OUTPUT STRING("Menu:")
    OUTPUT STRING("1 - Show current solution")
    OUTPUT STRING("2 - Next solution")
    OUTPUT STRING("3 - Previous solution")
    OUTPUT STRING("4 - Go to solution number")
    OUTPUT STRING("5 - Show ALL solutions")
    OUTPUT STRING("6 - Quit")

    choice <- INPUT(STRING("Enter choice:"))

    IF choice == STRING("1") THEN
        IF total > 0 THEN
            OUTPUT STRING("\\nShowing solution {index + 1} of {total}:\\n")
            print_board(solutions[index])
        ELSE
            OUTPUT STRING("\\nNo solutions exist for this N.\\n")
        ENDIF


    ELIF choice == STRING("2") THEN
        IF total > 0 THEN
            index <- (index + 1) MODULUS total
            OUTPUT STRING("\\nShowing solution {index + 1} of {total}:\\n")
            print_board(solutions[index])
        ELSE
            OUTPUT STRING("\\nNo solutions.\\n")
        ENDIF


    ELIF choice == STRING("3") THEN
        IF total > 0 THEN
            index <- (index - 1) MODULUS total
            OUTPUT STRING("\\nShowing solution {index + 1} of {total}:\\n")
            print_board(solutions[index])
        ELSE
            OUTPUT STRING("\\nNo solutions.\\n")
        ENDIF


    ELIF choice == STRING("4") THEN
        IF total > 0 THEN
            num <- INPUT(STRING("Enter solution number:"))

            IF CALL ISDIGIT(num) == BOOLEAN(TRUE) THEN
                num_int <- CALL TO_INTEGER(num)

                IF 1 <= num_int AND num_int <= total THEN
                    index <- num_int - 1
                    OUTPUT STRING("\\nShowing solution {index + 1} of {total}:\\n")
                    print_board(solutions[index])
                ELSE
                    OUTPUT STRING("\\nInvalid number.\\n")
                ENDIF

            ELSE
                OUTPUT STRING("\\nInvalid input.\\n")
            ENDIF

        ELSE
            OUTPUT STRING("\\nNo solutions.\\n")
        ENDIF


    ELIF choice == STRING("5") THEN
        print_all_solutions(solutions)


    ELIF choice == STRING("6") THEN
        OUTPUT STRING("\\nExiting program.\\n")
        loop <- BOOLEAN(FALSE)


    ELSE
        OUTPUT STRING("\\nInvalid option.\\n")
    ENDIF
ENDWHILE
