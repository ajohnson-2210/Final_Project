Find the first non-repeating character in a string

OUTPUT STRING("First Non Repeating Character In a String")

FUNCTION fnon_repeat ACCEPTS(s)
    chars <- CALL CHARACTERS_OF(s)
    counts <- LIST OF SIZE EQUALS LENGTH OF chars FILLED WITH 0
    i <- 0
    WHILE i < LENGTH OF chars DO
        j <- 0
        WHILE j < LENGTH OF chars DO
            IF chars[i] == chars[j] THEN
                counts[i] <- counts[i] + 1
            ENDIF
            j <- j + 1
        ENDWHILE
        i <- i + 1
    ENDWHILE

    i <- 0
    WHILE i < LENGTH OF chars DO
        IF counts[i] == 1 THEN
            RETURN STRING("String == {s} Result: {chars[i]}")
        ENDIF
        i <- i + 1
    ENDWHILE

    RETURN STRING("String == {s} Result: {CALL NULL()}")
ENDFUNCTION


OUTPUT STRING("{fnon_repeat(STRING(swiss))}")
OUTPUT STRING("{fnon_repeat(STRING(aabbccd))}")
OUTPUT STRING("{fnon_repeat(STRING(aabbcc))}")
